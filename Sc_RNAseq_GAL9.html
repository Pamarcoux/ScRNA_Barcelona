```{r}
Metadata_Bcell <- AllBcells@meta.data %>%
    select(orig.ident, type, Lymphomatype,Cluster_name,Response_type_total) %>%
    mutate(Sample_code = gsub("\\.", "_", orig.ident)) %>%
    mutate(cell_id = gsub("\\.", "_", rownames(.))) %>%
    mutate(
      Sample_code = ifelse(Sample_code == "FL5_PB1", "FL5_PB1_3", Sample_code),
      Sample_code = ifelse(Sample_code == "FL6_PB1", "FL6_PB1_2", Sample_code)
    ) |> 
  mutate(B_cell_type = case_when(
    # Cluster_name == "Circulating_B_Cells" ~ "Circulating_B_Tumor_B_Cells",
    Cluster_name %in% c("tFL4_PB1_2","DLBCL6_LN1_2","DLBCL2_LN1_2") ~ "Proliferating_Tumor_B_Cells",
    Cluster_name == "Healthy_B_Cells" ~ "Healthy_B_Cells",
    Cluster_name == "KI67+_B_Cells" ~ "Proliferating_Tumor_B_Cells",
    TRUE ~ "Non_Prolif_Tumor_B_Cells"
  )) |> 
  mutate(Lymphomatype = factor(Lymphomatype, levels = c("FL","tFL","DLBCL")),
         Sample_code_paper = Sample_code,
         Sample_code_paper = recode(Sample_code_paper, !!!Replacement_table),
         Sample_code = factor(Sample_code, levels = Sample_order),
         type = if_else(type == "PB", "PBMC", type)
)

  AllBcells <- AddMetaData(AllBcells, metadata = Metadata_Bcell$orig.ident, col.name = "orig.ident")
  AllBcells <- AddMetaData(AllBcells, metadata = Metadata_Bcell$Sample_code_paper, col.name = "Sample_code_paper")
  AllBcells <- AddMetaData(AllBcells, metadata = Metadata_Bcell$Sample_code, col.name = "Sample_code")
  AllBcells <- AddMetaData(AllBcells, metadata = Metadata_Bcell$B_cell_type, col.name = "B_cell_type")
  AllBcells <- AddMetaData(AllBcells, metadata = Metadata_Bcell$type, col.name = "Origin")
  AllBcells <- AddMetaData(AllBcells, metadata = Metadata_Bcell$Lymphomatype, col.name = "Disease")
```
